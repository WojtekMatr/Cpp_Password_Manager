cmake_minimum_required(VERSION 3.11)
project(TerminalApp)

# Ustawiamy standard C++ na 17 (wymagane przez FTXUI)
set(CMAKE_CXX_STANDARD 17)

# --- Magiczna część: Pobieranie FTXUI z internetu ---
include(FetchContent)
FetchContent_Declare(ftxui
  GIT_REPOSITORY https://github.com/ArthurSonzogni/FTXUI11:49 AM 1/29/2026
  GIT_TAG v5.0.0
)
FetchContent_MakeAvailable(ftxui)
# ----------------------------------------------------

add_executable(TerminalApp main.cpp)

# Linkowanie biblioteki do Twojego programu
target_link_libraries(TerminalApp PRIVATE ftxui::screen ftxui::dom ftxui::component)11:49 AM 1/29/2026
